import {
    ActionRowBuilder,
    ButtonBuilder,
    ButtonStyle,
    EmbedBuilder,
    MessageFlags,
    type ColorResolvable,
} from "discord.js";

import User from "../models/User";
import type { Command } from "../types/Command";

import {
    getShopPage,
} from "../services/shop/ShopService";

export async function executeShop(
    client: Parameters<Command["run"]>[0],
    interaction: Parameters<Command["run"]>[1],
    page = 0,
) {
    const user = await User.findOne({
        userId: interaction.user.id,
    });

    if (!user) {
        await interaction.reply({
            content:
                "Bạn chưa tạo tài khoản.\n" +
                "`/start` để bắt đầu hành trình cày cuốc của bạn.",
            flags: MessageFlags.Ephemeral,
        });

        return;
    }

    const result =
        getShopPage(
            client,
            user.level,
            user.pickaxe,
            page,
        );

    const description =
        result.pickaxes
            .map((pickaxe) => {
                const emoji =
                    client.appEmojis.get(
                        pickaxe.emoji,
                    ) ?? "";

                return [
                    `${emoji} **${pickaxe.name}**`,
                    pickaxe.description
                ].join("\n");
            })
            .join("\n\n");

    const embed =
        new EmbedBuilder()
            .setColor(
                user.color as ColorResolvable,
            )
            .setTitle(`${result.currentPickaxe ? client.appEmojis.get(result.currentPickaxe.emoji) : "⛏"} Pickaxe Shop`)
            .setDescription(
                [
                    `💰 Số dư: **$${user.balance.toLocaleString()}**`,
                    "",
                    description,
                ].join("\n"),
            )
            .setFooter({
                text:
                    `Trang ${result.page + 1}/${result.totalPages}`,
            });

    const components:
        ActionRowBuilder<ButtonBuilder>[] =
        [];
    
    const row =
            new ActionRowBuilder<ButtonBuilder>();
    for (
        const pickaxe of result.pickaxes
    ) {
        
        const buyable = user.balance >= pickaxe.price;
        const owned = user.unlocked_pickaxes.includes(pickaxe.id);

        row.addComponents(
            new ButtonBuilder()
                .setCustomId(
                    `shop:select:${pickaxe.id}`,
                )
                .setLabel(`${owned ? (pickaxe.id === result.currentPickaxe?.id ? "Đã trang bị" : "Trang bị") : `$${pickaxe.price}`}`)
                .setEmoji(
                    client.appEmojis.get(
                        pickaxe.emoji,
                    ) ?? "⛏️",
                )
                .setStyle(
                    owned
                        ? ButtonStyle.Success
                        : ButtonStyle.Primary,
                )
                .setDisabled(
                    !buyable || pickaxe.id === result.currentPickaxe?.id,
                ),
        );
    }
    components.push(row);

    const navigation =
        new ActionRowBuilder<ButtonBuilder>()
            .addComponents(
                new ButtonBuilder()
                    .setCustomId(
                        `shop:prev:${result.page}`,
                    )
                    .setLabel("Trang trước")
                    .setStyle(
                        ButtonStyle.Secondary,
                    )
                    .setDisabled(
                        result.page === 0,
                    ),

                new ButtonBuilder()
                    .setCustomId(
                        "shop:page",
                    )
                    .setLabel(
                        `${result.page + 1}/${result.totalPages}`,
                    )
                    .setStyle(
                        ButtonStyle.Secondary,
                    )
                    .setDisabled(true),

                new ButtonBuilder()
                    .setCustomId(
                        `shop:next:${result.page}`,
                    )
                    .setLabel("Trang kế")
                    .setStyle(
                        ButtonStyle.Primary,
                    )
                    .setDisabled(
                        result.page >=
                            result.totalPages - 1,
                    ),
            );

    components.push(navigation);

    if (interaction.isButton()) {
        await interaction.update({
            embeds: [embed],
            components,
        });

        return;
    }

    await interaction.reply({
        embeds: [embed],
        components,
    });
}

export default {
    name: "shop",
    description: "Cửa hàng bán cúp",

    run: async (
        client,
        interaction,
    ) => {
        await executeShop(
            client,
            interaction,
            0,
        );
    },
} satisfies Command;
